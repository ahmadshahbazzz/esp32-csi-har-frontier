\begin{document}\sansmath
\begin{tikzpicture}[
  font=\footnotesize, line width=0.5pt, >={Triangle[length=2mm,width=1.6mm]},
  blk/.style={draw=black,fill=white,rounded corners=1pt,align=center,inner sep=3pt},
  lbl/.style={font=\scriptsize,text=black},
  ]

% ============ left cluster: compute + on-chip peripherals ============
\node[blk,minimum width=26mm,minimum height=12mm] (c0) {Xtensa LX6\\core 0\\\scriptsize 240\,MHz};
\node[blk,minimum width=26mm,minimum height=12mm,right=4mm of c0] (c1) {Xtensa LX6\\core 1\\\scriptsize 240\,MHz};

\node[blk,minimum width=26mm,minimum height=11mm,below=10mm of c0.south,anchor=north] (rom) {ROM 448\,kB\\\scriptsize boot};
\node[blk,minimum width=26mm,minimum height=11mm,below=10mm of c1.south,anchor=north] (rtc) {RTC SRAM\\16\,kB};

\node[blk,dashed,minimum width=26mm,minimum height=11mm,below=6mm of rom.south,anchor=north] (rf) {Wi-Fi / BT radio\\\scriptsize off in measurement};
\node[blk,minimum width=26mm,minimum height=11mm,below=6mm of rtc.south,anchor=north] (peri) {peripherals\\\scriptsize GPIO, ADC, UART, SPI};

% ---- system bus (rail between cores and the lower blocks) ----
\coordinate (busY) at ($(c0.south)!0.5!(rom.north)$);
\coordinate (busL) at (c0.west|-busY);
\coordinate (busR) at (c1.east|-busY);
\draw[line width=0.9pt] (busL) -- (busR);
\foreach \n in {c0,c1}{\draw (\n.south) -- (\n.south|-busY);}
\foreach \n in {rom,rtc}{\draw (\n.north) -- (\n.north|-busY);}
% bus junction dots
\foreach \n in {c0,c1}{\fill (\n.south|-busY) circle (0.5mm);}
% bus label in the clear gap between the cores column and the SRAM (set after SRAM)
% radio + peripherals hang below the memory row (short connectors, no crossing)
\draw (rf.north)   -- (rf.north|-rom.south);
\draw (peri.north) -- (peri.north|-rtc.south);

% ============ right: internal SRAM, full SoC height ============
\node[blk,minimum width=42mm,minimum height=45mm,anchor=north west] (sram) at ([xshift=13mm]c1.north east) {};
\node[lbl,anchor=north,font=\scriptsize\bfseries] at ([yshift=-1.6mm]sram.north) {Internal SRAM\; 520\,kB};
% usable arena, drawn as a hatched sub-region (wireframe)
\node[draw=black,pattern=north east lines,pattern color=black!35,minimum width=36mm,minimum height=15mm,anchor=north]
  (arena) at ([yshift=-7mm]sram.north) {};
\node[align=center,font=\scriptsize] at (arena) {$\approx$150\,kB largest\\contiguous block\\\textbf{= tensor arena}};
\node[lbl,align=center,font=\scriptsize,anchor=north] at ([yshift=-3mm]arena.south)
  {remaining $\approx$370\,kB:\\static data, stacks,\\Wi-Fi/BT buffers,\\fragmentation\\(non-contiguous)};
% memory bus into the SRAM, with the bus label in this clear gap
\draw (busR) -- (sram.west|-busY);
\node[lbl,font=\scriptsize\itshape,anchor=south] at ($(busR)!0.5!(sram.west|-busY)+(0,0.8mm)$) {system bus};

% ============ SoC boundary ============
\begin{scope}[on background layer]
\node[draw=black,line width=0.9pt,rounded corners=2pt,fit=(c0)(c1)(rom)(rtc)(rf)(peri)(sram),inner sep=5mm] (soc) {};
\end{scope}
\node[anchor=south west,font=\bfseries] at ([xshift=1mm,yshift=0.5mm]soc.north west) {ESP32-D0WD-V3 SoC};
\node[anchor=south east,align=right,font=\scriptsize\bfseries] at ([xshift=-1mm,yshift=0.5mm]soc.north east)
  {NO PSRAM \quad scalar cores (no SIMD)};

% ============ external flash (on the WROOM-32 module, outside the SoC) ============
\node[blk,dashed,minimum width=58mm,minimum height=12mm,below=10mm of soc.south] (flash)
  {\textbf{External 4\,MB SPI flash} \; \scriptsize(on the ESP32-WROOM-32 module)\\\scriptsize holds int8 model weights, memory-mapped, not in SRAM};
\draw[<->] (soc.south) -- (flash.north) node[midway,right=1mm,lbl]{QSPI};

\end{tikzpicture}
