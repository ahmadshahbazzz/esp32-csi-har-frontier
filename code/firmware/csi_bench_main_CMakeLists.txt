# Frontier firmware: one tiny int8 model at a time.
# Regenerate model_tiny.cc per model with:
#   python esp32_firmware/tflite_to_c.py <model_int8.tflite> esp32_firmware/csi_bench/main/model_tiny.cc

# Item #2: only pull in the WiFi component stack when the WIFI_ON experiment build is
# requested (otherwise the default build has WiFi disabled and must not require it).
set(extra_reqs "")
if(DEFINED ENV{WIFI_ON})
    list(APPEND extra_reqs esp_wifi nvs_flash esp_netif esp_event)
endif()
# Item #8 energy: the ENERGY build drives a marker GPIO, which needs the driver component.
if(DEFINED ENV{ENERGY})
    list(APPEND extra_reqs driver)
endif()
# Major #1 live pipeline: LIVE=1 brings up the WiFi station + CSI RX + ping, so it needs
# the radio stack and lwip (esp_ping lives in lwip).
if(DEFINED ENV{LIVE})
    list(APPEND extra_reqs esp_wifi nvs_flash esp_netif esp_event lwip)
endif()

idf_component_register(
    SRCS "main.cc" "model_tiny.cc"
    INCLUDE_DIRS "."
    REQUIRES esp-tflite-micro esp_timer ${extra_reqs}
)

# Item #11 replay demo: build with REPLAY_DEMO=1 in the environment to compile the
# stored-window classification path (requires main/replay_windows.h) instead of the
# latency loop. Normal latency builds leave REPLAY_DEMO unset and are unaffected.
if(DEFINED ENV{REPLAY_DEMO})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE REPLAY_DEMO=1)
endif()
# Item #2 WiFi experiment: WIFI_ON=1 brings up the radio stack before measuring RAM.
if(DEFINED ENV{WIFI_ON})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE WIFI_ON=1)
endif()
# Item #25 arena ablation: ARENA_BYTES=<n> forces a fixed tensor-arena size.
if(DEFINED ENV{ARENA_BYTES})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE ARENA_BYTES=$ENV{ARENA_BYTES})
endif()
# Item #16 per-layer arena: RECORD_ALLOC=1 uses the RecordingMicroInterpreter to print
# the tensor-arena allocation breakdown, then returns (no latency loop).
if(DEFINED ENV{RECORD_ALLOC})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE RECORD_ALLOC=1)
endif()
# Items #10/#11 continuous run: CONTINUOUS=1 runs back-to-back inference for a long
# time, sampling heap and latency to check runtime stability / fragmentation.
if(DEFINED ENV{CONTINUOUS})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE CONTINUOUS=1)
endif()
# Item #8 energy: ENERGY=1 drives a marker GPIO HIGH during each Invoke() so an
# external INA219+ESP8266 reader can separate active from idle power. MARKER_PIN
# optionally overrides the default GPIO25.
if(DEFINED ENV{ENERGY})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE ENERGY=1)
    if(DEFINED ENV{MARKER_PIN})
        target_compile_definitions(${COMPONENT_LIB} PRIVATE MARKER_PIN=$ENV{MARKER_PIN})
    endif()
endif()
# Major #1 live pipeline: LIVE=1 compiles the WiFi-STA + CSI-RX + windowed-inference path.
if(DEFINED ENV{LIVE})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE LIVE=1)
    if(DEFINED ENV{LIVE_MINUTES})
        target_compile_definitions(${COMPONENT_LIB} PRIVATE LIVE_MINUTES=$ENV{LIVE_MINUTES})
    endif()
endif()
